# GRID SEARCH FOR OPTIMAL KNOTS (tau_M, tau_A, tau_C)
# Separate RSS-minimising search for marking, admin, CCA
# over their observed support (excluding min and max)
# =============================================================================

find_knot <- function(data, var, other_vars, dv, weight_var,
                     probs = c(.25, .35, .45, .5, .55, .65, .75)) {
    # Uses lm() not svyglm() for speed: RSS grid search over many candidate knots.
  # Cluster-robust SEs do not materially affect knot location.
  # Final piecewise model uses svyglm() with cluster-robust SEs throughout.
  all_support <- sort(unique(data[[var]]))
  quantile_bounds <- quantile(data[[var]], probs = probs, na.rm = TRUE)
  support <- all_support[all_support >= min(quantile_bounds) &
                         all_support <= max(quantile_bounds)]
  if (length(support) == 0) support <- all_support  # fallback

  rss_vec <- numeric(length(support))

  for (i in seq_along(support)) {
    tau         <- support[i]
    data$hinge  <- pmax(0, data[[var]] - tau)

    fmla <- as.formula(paste(
      dv, "~", var, "+ hinge +",
      paste(other_vars, collapse = " + ")
    ))

    fit        <- lm(fmla, data = data, weights = data[[weight_var]])
    rss_vec[i] <- sum(fit$residuals^2)
  }

  best_tau <- support[which.min(rss_vec)]
  return(best_tau)
}

# (A) Quadratic model (Model 2): predicted change in stress over an interval
#     [a, b] of marking hours, holding all else fixed.
#     Delta = beta_linear*(b-a) + beta_quad*(b^2 - a^2)
# -----------------------------------------------------------------------------
quad_delta <- function(coef_vec, lin_name, quad_name, a, b) {
  beta_lin  <- coef_vec[lin_name]
  beta_quad <- coef_vec[quad_name]
  as.numeric(beta_lin * (b - a) + beta_quad * (b^2 - a^2))
}

plot_rss <- function(data, var, other_vars, dv, weight_var, title) {
  support <- sort(unique(data[[var]]))
  support <- support[support > min(support) & support < max(support)]
  rss_vec <- numeric(length(support))
  for (i in seq_along(support)) {
    data$hinge <- pmax(0, data[[var]] - support[i])
    fmla <- as.formula(paste(dv, "~", var, "+ hinge +",
                             paste(other_vars, collapse = " + ")))
    fit <- lm(fmla, data = data, weights = data[[weight_var]])
    rss_vec[i] <- sum(fit$residuals^2)
  }
  plot(support, rss_vec, type = "l", main = title,
       xlab = "Candidate knot (hours)", ylab = "RSS")
  abline(v = support[which.min(rss_vec)], col = "red", lty = 2)
  cat(title, "— optimal knot:", support[which.min(rss_vec)], "\n")
}

compute_rss <- function(data, focal_var, other_vars, dv, weight_var, tau) {
  data$hinge <- pmax(0, data[[focal_var]] - tau)
  fmla <- as.formula(paste(dv, "~", focal_var, "+ hinge +",
                           paste(other_vars, collapse = " + ")))
  fit <- lm(fmla, data = data, weights = data[[weight_var]])
  sum(fit$residuals^2)
}

rss_over_support <- function(data, focal_var, other_vars, dv, weight_var) {
  support <- sort(unique(data[[focal_var]]))
  support <- support[support > min(support) & support < max(support)]
  rss_vec <- sapply(support, function(tau)
    compute_rss(data, focal_var, other_vars, dv, weight_var, tau))
  tibble(tau = support, rss = rss_vec)
}

  lbl_res   <- paste0("IQR-restricted min (\u03C4=", tau_res, "h)")

# STEP 2: Load pre-exclusion sample from raw SPSS
# Pull same variables, apply same cleaning, but skip 84h exclusion
# -----------------------------------------------------------------------------
df_raw <- read_sav("ttgintt4.sav")

df_pre_raw <- df_raw %>%
  filter(CNTRY == "SGP") %>%
  # Replace missing codes
  mutate(across(c(TT4G16A,TT4G16B,TT4G16C,TT4G16D,TT4G16E,
                  TT4G16F,TT4G16G,TT4G16H,TT4G16I,TT4G16J),
                ~ifelse(. %in% c(998, 999), NA, .)),
         T4WLOADT = ifelse(T4WLOADT %in% c(9998, 9999), NA, T4WLOADT),
         across(c(TT4G01,T4TYEXPTT,T4TYEXPTS,TT4G04,
                  TT4G09,T4TEMPWH,T4TNSCH,TT4G64D,TT4G64H),
                ~ifelse(. %in% c(8, 9), NA, .)))

# Rotation form filter
rotation_vars <- c("T4WLOADT","TT4G16A","TT4G16B","TT4G16C","TT4G16D","TT4G16E",
                   "TT4G16F","TT4G16G","TT4G16H","TT4G16I","TT4G16J")
rotation_absent <- rowSums(!is.na(df_pre_raw[rotation_vars])) == 0
df_pre_raw <- df_pre_raw[!rotation_absent, ]

# Listwise deletion
analysis_vars <- c("T4WLOADT","TT4G16A","TT4G16B","TT4G16C","TT4G16D","TT4G16E",
                   "TT4G16F","TT4G16G","TT4G16H","TT4G16I","TT4G16J",
                   "TT4G01","T4TYEXPTT","T4TYEXPTS","TT4G04",
                   "TT4G09","T4TEMPWH","T4TNSCH","TT4G64D","TT4G64H")
df_pre_raw <- df_pre_raw %>% filter(complete.cases(across(all_of(analysis_vars))))
cat("Pre-exclusion sample (no 84h filter): N =", nrow(df_pre_raw), "\n")
## Pre-exclusion sample (no 84h filter): N = 1768
## Expected: 1,768

# -----------------------------------------------------------------------------
# STEP 3: Rename and recode — same as Python Cell 8
# -----------------------------------------------------------------------------
df_pre <- df_pre_raw %>%
  rename(
    workload_stress_scale        = T4WLOADT,
    hrs_lesson_planning          = TT4G16A,
    hrs_teamwork_collaboration   = TT4G16B,
    hrs_marking_correcting       = TT4G16C,
    hrs_student_counselling      = TT4G16D,
    hrs_school_management        = TT4G16E,
    hrs_general_admin            = TT4G16F,
    hrs_professional_development = TT4G16G,
    hrs_parent_communication     = TT4G16H,
    hrs_cca_extracurricular      = TT4G16I,
    hrs_other_tasks              = TT4G16J,
    teaching_exp_total           = T4TYEXPTT,
    teaching_exp_current_school  = T4TYEXPTS,
    sampling_weight              = TCHWGT,
    school_id                    = IDSCHOOL
  ) %>%
  mutate(
    total_nonteaching = rowSums(across(c(hrs_lesson_planning,
                                          hrs_teamwork_collaboration,
                                          hrs_marking_correcting,
                                          hrs_student_counselling,
                                          hrs_school_management,
                                          hrs_general_admin,
                                          hrs_professional_development,
                                          hrs_parent_communication,
                                          hrs_cca_extracurricular,
                                          hrs_other_tasks))),
    gender = case_when(TT4G01 == 1 ~ 1, TT4G01 == 2 ~ 0, TRUE ~ NA_real_),
    teacher_qualification = case_when(
      TT4G04 %in% c(1,2) ~ 0, TT4G04 %in% c(3,4,5) ~ 1, TRUE ~ NA_real_),
    contract_type = case_when(
      TT4G09 == 1 ~ 1, TT4G09 %in% c(2,3) ~ 0, TRUE ~ NA_real_),
    employment_hours = case_when(
      T4TEMPWH == 1 ~ 1, T4TEMPWH == 2 ~ 0, TRUE ~ NA_real_),
    multi_school = case_when(
      T4TNSCH == 0 ~ 0, T4TNSCH %in% c(1,2) ~ 1, TRUE ~ NA_real_),
    collab_culture = case_when(
      TT4G64D %in% c(1,2) ~ 0, TT4G64D %in% c(3,4) ~ 1, TRUE ~ NA_real_),
    peer_support = case_when(
      TT4G64H %in% c(1,2) ~ 0, TT4G64H %in% c(3,4) ~ 1, TRUE ~ NA_real_),
    gender                = factor(gender),
    teacher_qualification = factor(teacher_qualification),
    contract_type         = factor(contract_type),
    employment_hours      = factor(employment_hours),
    multi_school          = factor(multi_school),
    collab_culture        = factor(collab_culture),
    peer_support          = factor(peer_support),
    exp_total  = factor(teaching_exp_total, levels = 1:4,
                        labels = c("le5yr","6to10yr","11to20yr","gt20yr")),
    exp_school = factor(teaching_exp_current_school, levels = 1:4,
                        labels = c("le5yr","6to10yr","11to20yr","gt20yr")),
    marking_sq = hrs_marking_correcting^2,
    admin_sq   = hrs_general_admin^2,
    cca_sq     = hrs_cca_extracurricular^2
  )
